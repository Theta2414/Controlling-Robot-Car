\documentclass{standalone}
\usepackage{style}
\usetikzlibrary{arrows.meta, positioning, calc}

\tikzset{
    block/.style={
        draw,
        rounded corners=4pt,
        minimum width=2.5cm,
        minimum height=1.0cm,
        align=center,
        fill=white,
        semithick
    },
    power flow/.style={
        dashed,
        -{Stealth[scale=1.0]},
        semithick
    },
    bi-power flow/.style={
        dashed,
        {Stealth[scale=1.0]}-{Stealth[scale=1.0]},
        semithick
    },
    data flow/.style={
        -{Stealth[scale=1.0]},
        semithick
    },
    control flow/.style={
        -{Triangle[open, scale=1.0]},
        semithick
    }
}

\begin{document}
\begin{tikzpicture}[font=\small]
    % --- KHỐI PHÁT ---
    % Khung bao khối phát (Rộng 8.4cm, Cao 8.7cm)
    \draw[thick] (0, 0) rectangle (8.4, -8.7);
    \node[anchor=north] at (4.2, -8.8) {\textbf{Khối phát}};

    % Các khối bên trong Khối phát
    \node[block] (Nguon_TX) at (2.0, -3.6) {Nguồn \qty{7.4}{\volt}};
    \node[block] (STM_TX) at (6.0, -3.6) {STM32F103C8T6};
    \node[block] (MPU) at (6.0, -1.3) {MPU6050};
    \node[block] (HC_TX) at (6.0, -5.9) {HC-05};

    % Đường nối Khối phát (Dài 4cm tâm-tâm, khoảng trống 1.5cm)
    \draw[power flow] (Nguon_TX) -- (STM_TX) node[midway, above] {Hạ áp};
    
    % MPU6050 <-> STM32
    \draw[bi-power flow] ($(STM_TX.north)+(-0.5, 0)$) -- ($(MPU.south)+(-0.5, 0)$);
    \draw[{Stealth[scale=1.0]}-{Triangle[open, scale=1.0]}, semithick] ($(STM_TX.north)+(0.5, 0)$) -- ($(MPU.south)+(0.5, 0)$) node[midway, right] {I2C};

    % HC-05 <-> STM32
    \draw[bi-power flow] ($(STM_TX.south)+(-0.5, 0)$) -- ($(HC_TX.north)+(-0.5, 0)$);
    \draw[{Stealth[scale=1.0]}-{Triangle[open, scale=1.0]}, semithick] ($(STM_TX.south)+(0.5, 0)$) -- ($(HC_TX.north)+(0.5, 0)$) node[midway, right] {UART};

    % Chú thích (Legend) - Căn gọn bên góc trái
    \begin{scope}[shift={(0.4, -6.5)}]
        \draw[data flow] (0, 0) -- (1.2, 0) node[right] {Luồng dữ liệu};
        \draw[control flow] (0, -0.6) -- (1.2, -0.6) node[right] {Luồng điều khiển};
        \draw[power flow] (0, -1.2) -- (1.2, -1.2) node[right] {Đường cấp nguồn};
    \end{scope}

    % --- KHỐI THU ---
    % Khung bao khối thu (Rộng 8.4cm, Cao 8.7cm)
    \draw[thick] (9.6, 0) rectangle (18.0, -8.7);
    \node[anchor=north] at (13.8, -8.8) {\textbf{Khối thu}};

    % Các khối bên trong Khối thu
    \node[block] (HC_RX) at (12.0, -1.3) {HC-05};
    \node[block] (STM_RX) at (12.0, -3.6) {STM32F103C8T6};
    \node[block] (TB) at (12.0, -5.9) {Module Motor\\Driver TB6612};
    \node[block] (Motor) at (12.0, -7.8) {Động cơ DC có\\chổi than};
    \node[block] (Nguon_RX) at (16.0, -3.6) {Nguồn \qty{7.4}{\volt}};

    % Đường nối Khối thu
    % HC-05 <-> STM32
    \draw[bi-power flow] ($(HC_RX.south)+(-0.5, 0)$) -- ($(STM_RX.north)+(-0.5, 0)$);
    \draw[{Triangle[open, scale=1.0]}-{Stealth[scale=1.0]}, semithick] ($(HC_RX.south)+(0.5, 0)$) -- ($(STM_RX.north)+(0.5, 0)$) node[midway, right] {UART};

    % STM32 <-> TB6612
    \draw[bi-power flow] ($(STM_RX.south)+(-0.5, 0)$) -- ($(TB.north)+(-0.5, 0)$);
    \draw[control flow] ($(STM_RX.south)+(0.5, 0)$) -- ($(TB.north)+(0.5, 0)$) node[midway, right] {PWM \& GPIO};

    % TB6612 -> Motor
    \draw[power flow] (TB.south) -- (Motor.north);

    % Nguồn -> STM32 / TB6612 (Dài 4cm tâm-tâm, khoảng trống 1.5cm)
    \draw[power flow] (Nguon_RX.west) -- (STM_RX.east) node[midway, above] {Hạ áp};
    \draw[power flow] (Nguon_RX.south) |- (TB.east);

    % --- ĐƯỜNG TRUYỀN BLUETOOTH SPP ---
    % Căn giữa đường truyền nối tiếp tại x = 9.0 (nằm giữa x=8.4 và x=9.6)
    \draw[data flow] (HC_TX.east) -- (9.0, -5.9) |- (HC_RX.west) node[pos=0.25, fill=white, align=center, inner sep=1pt] {\tiny Bluetooth\\\tiny SPP};

\end{tikzpicture}
\end{document}
